CREATE TABLE users (
    user_id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    email VARCHAR(150) UNIQUE NOT NULL,
    role ENUM('student') DEFAULT 'student',
    college VARCHAR(200) NOT NULL,
    password VARCHAR(255) NOT NULL
);

CREATE TABLE organizers (
    org_id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    email VARCHAR(150) UNIQUE NOT NULL,
    role ENUM('organizer') DEFAULT 'organizer',
    company_name_or_college_name VARCHAR(200) NOT NULL,
    password VARCHAR(255) NOT NULL
);

CREATE TABLE hackathons (
    hackathon_id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(200) NOT NULL,
    type ENUM('online', 'offline') NOT NULL,
    venue VARCHAR(200),
    start_date DATE NOT NULL,
    end_date DATE NOT NULL,
    description TEXT NOT NULL,
    content TEXT NOT NULL,
    file_path VARCHAR(255) NOT NULL,
    org_id INT NOT NULL,
    FOREIGN KEY (org_id) REFERENCES organizers(org_id) ON DELETE CASCADE,
    is_certificates_issued BOOLEAN DEFAULT FALSE
);

CREATE TABLE participants (
    user_id INT NOT NULL,
    hackathon_id INT NOT NULL,
    PRIMARY KEY (user_id, hackathon_id),
    current_round INT DEFAULT 1,
    FOREIGN KEY (user_id) REFERENCES users(user_id) ON DELETE CASCADE,
    FOREIGN KEY (hackathon_id) REFERENCES hackathons(hackathon_id) ON DELETE CASCADE
);

CREATE TABLE teams (
    team_id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(150) NOT NULL,
    hackathon_id INT NOT NULL,
    team_leader_id INT NOT NULL,
    FOREIGN KEY (hackathon_id) REFERENCES hackathons(hackathon_id) ON DELETE CASCADE,
    FOREIGN KEY (team_leader_id) REFERENCES users(user_id) ON DELETE CASCADE
);

CREATE TABLE team_members (
    team_id INT NOT NULL,
    user_id INT NOT NULL,
    PRIMARY KEY (team_id, user_id),
    FOREIGN KEY (team_id) REFERENCES teams(team_id) ON DELETE CASCADE,
    FOREIGN KEY (user_id) REFERENCES users(user_id) ON DELETE CASCADE
);

 

CREATE TABLE messages (
    message_id INT AUTO_INCREMENT PRIMARY KEY,
    team_id INT NOT NULL,
    sender_id INT NOT NULL,
    message TEXT NOT NULL,
    timestamp TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (team_id) REFERENCES teams(team_id) ON DELETE CASCADE,
    FOREIGN KEY (sender_id) REFERENCES users(user_id) ON DELETE CASCADE
);

DELIMITER $$

CREATE TRIGGER before_message_insert
BEFORE INSERT ON messages
FOR EACH ROW
BEGIN
    IF NOT EXISTS (
        SELECT 1 FROM team_members 
        WHERE team_members.team_id = NEW.team_id 
        AND team_members.user_id = NEW.sender_id
    ) THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'Sender must be a member of the team';
    END IF;
END $$

CREATE TABLE rounds (
    id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    type ENUM('quiztype', 'documenttype') NOT NULL,
    hackathon_id INT NOT NULL,
    FOREIGN KEY (hackathon_id) REFERENCES hackathons(hackathon_id) ON DELETE CASCADE
);

CREATE TABLE quiz_type (
    id INT AUTO_INCREMENT PRIMARY KEY,
    round_id INT NOT NULL,
    hackathon_id INT NOT NULL,
    question TEXT NOT NULL,
    op1 VARCHAR(200) NOT NULL,
    op2 VARCHAR(200) NOT NULL,
    op3 VARCHAR(200) NOT NULL,
    op4 VARCHAR(200) NOT NULL,
    correct_answer VARCHAR(200) NOT NULL,
    FOREIGN KEY (round_id) REFERENCES rounds(id) ON DELETE CASCADE,
    FOREIGN KEY (hackathon_id) REFERENCES hackathons(hackathon_id) ON DELETE CASCADE
);

CREATE TABLE document_type (
    id INT AUTO_INCREMENT PRIMARY KEY,
    round_id INT NOT NULL,
    hackathon_id INT NOT NULL,
    statement TEXT NOT NULL,
    FOREIGN KEY (round_id) REFERENCES rounds(id) ON DELETE CASCADE,
    FOREIGN KEY (hackathon_id) REFERENCES hackathons(hackathon_id) ON DELETE CASCADE
);

CREATE TABLE quiz_type_score (
    user_id INT NOT NULL,
    hackathon_id INT NOT NULL,
    round_id INT NOT NULL,
    score INT NOT NULL,
    PRIMARY KEY (user_id, hackathon_id, round_id),
    FOREIGN KEY (user_id) REFERENCES users(user_id) ON DELETE CASCADE,
    FOREIGN KEY (hackathon_id) REFERENCES hackathons(hackathon_id) ON DELETE CASCADE,
    FOREIGN KEY (round_id) REFERENCES rounds(id) ON DELETE CASCADE
);

CREATE TABLE document_type_submissions (
    user_id INT NOT NULL,
    hackathon_id INT NOT NULL,
    round_id INT NOT NULL,
    filepath TEXT NOT NULL,
    PRIMARY KEY (user_id, hackathon_id, round_id),
    FOREIGN KEY (user_id) REFERENCES users(user_id) ON DELETE CASCADE,
    FOREIGN KEY (hackathon_id) REFERENCES hackathons(hackathon_id) ON DELETE CASCADE,
    FOREIGN KEY (round_id) REFERENCES rounds(id) ON DELETE CASCADE
);

-- New table for document submission scores
CREATE TABLE document_submission_scores (
    user_id INT NOT NULL,
    hackathon_id INT NOT NULL,
    round_id INT NOT NULL,
    score DECIMAL(5,2) NOT NULL,
    feedback TEXT,
    evaluated_by INT   NULL,
    evaluated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    PRIMARY KEY (user_id, hackathon_id, round_id),
    FOREIGN KEY (user_id) REFERENCES users(user_id) ON DELETE CASCADE,
    FOREIGN KEY (hackathon_id) REFERENCES hackathons(hackathon_id) ON DELETE CASCADE,
    FOREIGN KEY (round_id) REFERENCES rounds(id) ON DELETE CASCADE,
    FOREIGN KEY (evaluated_by) REFERENCES organizers(org_id) ON DELETE CASCADE
);

-- New table for round winners (simplified without ranks)
CREATE TABLE round_winners (
    round_id INT NOT NULL,
    hackathon_id INT NOT NULL,
    user_id INT NOT NULL,
    team_id INT,
    PRIMARY KEY (round_id, hackathon_id, user_id),
    FOREIGN KEY (round_id) REFERENCES rounds(id) ON DELETE CASCADE,
    FOREIGN KEY (hackathon_id) REFERENCES hackathons(hackathon_id) ON DELETE CASCADE,
    FOREIGN KEY (user_id) REFERENCES users(user_id) ON DELETE CASCADE,
    FOREIGN KEY (team_id) REFERENCES teams(team_id) ON DELETE SET NULL
);

CREATE TABLE winners (
    user_id INT NOT NULL,
    hackathon_id INT NOT NULL,
    PRIMARY KEY (user_id, hackathon_id),
    FOREIGN KEY (user_id) REFERENCES users(user_id) ON DELETE CASCADE,
    FOREIGN KEY (hackathon_id) REFERENCES hackathons(hackathon_id) ON DELETE CASCADE
);

CREATE TABLE team_requests (
    request_id INT AUTO_INCREMENT PRIMARY KEY,
    team_id INT NOT NULL,
    sender_id INT NOT NULL,
    receiver_id INT NOT NULL,
    status ENUM('pending', 'accepted', 'declined') DEFAULT 'pending',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (team_id) REFERENCES teams(team_id) ON DELETE CASCADE,
    FOREIGN KEY (sender_id) REFERENCES users(user_id) ON DELETE CASCADE,
    FOREIGN KEY (receiver_id) REFERENCES users(user_id) ON DELETE CASCADE
);
